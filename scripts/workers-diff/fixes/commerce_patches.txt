// fix-cors-methods · fix-preflight-session: CORS 머리글 (서버가 처리하는 PUT · DELETE 포함)
function corsHeaders(h){
	h.set('Access-Control-Allow-Origin', '*')
	h.set('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS')
	h.set('Access-Control-Allow-Headers', 'Content-Type, Content-Encoding, Authorization, X-Requested-With')
	return h
}

// 글 → 평범한 JSON 객체 (배열 · null · 글 · 수 · 깨진 JSON 이면 null)
function appJson(text){
	try{
		var o = JSON.parse(text)
		return o && typeof o == "object" && !Array.isArray(o) ? o : null
	}catch(err){
		return null
	}
}

function pageLink(href, origin){
	try{
		var url

			url = new URL(href)
		} catch (e) {
			url = new URL(href, origin || "https://commerce.logis.center")

// fix-revoke-owner · fix-app-invite: 팀장 = 팀 행(id == to == 팀)의 from. 1089 가 찾은 admin 이 있으면 그것, 없으면(created_at 이 없는 요청) 읽는다

		return admin.from

// fix-app-invite: 멤버 행 — 읽는 곳(1567)의 꼴: type 'member', from = 멤버 주소, to = 팀, cc = 준 사람, bcc = 그 페이지 행의 bcc, ref = 페이지 id.
// 페이지는 그 팀의 zone items 의 type 'pages' 행만 (없으면 건너뜀). id = hashId(멤버 + 팀 + bcc)
async function memberStatements(env, zone, member, team, granter, refs, data, now){
	var db = env[`commerce_logis_${zone}_items`]
	var statements = []
	var rows = []
	for(var i = 0; i < refs.length; i++){
		var { results } = await db.prepare(`SELECT "id", "bcc" FROM items WHERE "type" = 'pages' AND "id" = ?1 AND "to" = ?2 LIMIT 1`).bind(refs[i], team).all()
		if(!results.length || !results[0].bcc){
			continue
		}
		var page = results[0]
		var id = hashId(member + team + page.bcc)
		if(rows.some(row => row.id == id)){
			continue
		}
		statements.push(db.prepare(`
			INSERT INTO items ("id", "type", "from", "to", "cc", "bcc", "ref", "data", "created_at", "updated_at")
			VALUES (?1, 'member', ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?8) ON CONFLICT (id) DO UPDATE SET
				"cc" = EXCLUDED."cc",
				"ref" = EXCLUDED."ref",
				"data" = EXCLUDED."data",
				"updated_at" = EXCLUDED."updated_at"
		`).bind(id, member, team, granter, page.bcc, page.id, data, now))
		rows.push({ id : id, table : 'items', type : 'member', from : member, to : team, cc : granter, bcc : page.bcc, ref : page.id })
	}
	return { statements : statements, rows : rows }
}

// fix-app-relay: WebRTC 신호 중계 — 데스크톱(main.ts 4856)이 POST /relay/{hash} 로 offer 를 올리고 모바일(main.ts 1260)이 GET 으로 읽는다.
// POST 는 그 hash 의 세션이 있어야 한다(질의 token 이 S3 hash/{hash} 의 token 과 같아야). GET 은 10분 안의 것만, 아니면 404 {}
async function relayFetch(request, env, s3, hash){
	var headers = corsHeaders(new Headers())
	headers.set('Content-Type', 'application/json')
	var key = `https://${env.aws_bucket}.s3.${env.aws_region}.amazonaws.com/relay/${hash}`
	if(request.method == 'POST'){
		var token = new URL(request.url).searchParams.get('token')
		if(!token){
			return new Response('{}', { status : 401, headers : headers })
		}
		var head = await s3.fetch(`https://${env.aws_bucket}.s3.${env.aws_region}.amazonaws.com/hash/${hash}`, { method : 'HEAD' })
		if(!head.ok || head.headers.get('x-amz-meta-token') != token){
			return new Response('{}', { status : 401, headers : headers })
		}
		var text = await request.text()
		var payload = appJson(text)
		if(!payload || text.length > 65536){
			return new Response('{}', { status : 400, headers : headers })
		}
		await s3.fetch(key, {
			method : 'PUT',
			headers : { 'Content-Type' : 'application/json', 'x-amz-meta-created_at' : String(Date.now()) },
			body : JSON.stringify(payload)
		})
		return new Response(JSON.stringify({ success : true }), { headers : headers })
	}
	if(request.method == 'GET'){
		var res = await s3.fetch(key, { method : 'GET' })
		if(res.ok && Date.now() - Number(res.headers.get('x-amz-meta-created_at')) < 600000){
			return new Response(await res.text(), { headers : headers })
		}
		return new Response('{}', { status : 404, headers : headers })
	}
	return new Response('{}', { status : 405, headers : headers })
}
